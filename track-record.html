⚠️ 本ページに関する重要なお知らせ

import pandas as pd
df = pd.read_csv("https://marketwatch-jp.com/signals-log.csv")

# 曜日別勝率

df.groupby("発火_曜日")["win"].mean()

# 銘柄 × 曜日 マトリクス
df.pivot_table(values="win", index="ticker", columns="発火_曜日", aggfunc="mean")

df["fired_at"] = pd.to_datetime(df["fired_at"])
